# AegisGuard FIM - Enterprise File Integrity Monitoring Daemon
# Engine: Python 3.10+ | SHA-256 Cryptographic Verification | Watchdog Filesystem

import os
import sys
import time
import hashlib
import sqlite3
import datetime
from pathlib import Path
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

# Database Initialization
DB_PATH = "fim_baseline.db"

def get_db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def init_database():
    """Create baseline and audit log tables."""
    conn = get_db()
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS file_baselines (
            filepath TEXT PRIMARY KEY,
            sha256 TEXT NOT NULL,
            permissions TEXT NOT NULL,
            file_size INTEGER NOT NULL,
            last_modified TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS security_alerts (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            event_type TEXT NOT NULL,
            severity TEXT NOT NULL,
            filepath TEXT NOT NULL,
            old_hash TEXT,
            new_hash TEXT,
            details TEXT
        )
    """)
    conn.commit()
    conn.close()
    print("[+] FIM Database Initialized.")

def calculate_sha256(filepath: str, block_size: int = 65536) -> str:
    """Calculate cryptographic SHA-256 hash in blocks to handle large files."""
    sha256 = hashlib.sha256()
    try:
        with open(filepath, "rb") as f:
            for block in iter(lambda: f.read(block_size), b""):
                sha256.update(block)
        return sha256.hexdigest()
    except Exception as e:
        return f"ERROR: {str(e)}"

def establish_baseline(directory: str):
    """Scan directory recursively and register initial SHA-256 baselines."""
    conn = get_db()
    cursor = conn.cursor()
    print(f"[*] Establishing cryptographic baseline for directory: {directory}")
    
    count = 0
    for root, _, files in os.walk(directory):
        for file in files:
            full_path = os.path.abspath(os.path.join(root, file))
            file_hash = calculate_sha256(full_path)
            stat = os.stat(full_path)
            perms = oct(stat.st_mode)[-3:]
            size = stat.st_size
            mtime = datetime.datetime.fromtimestamp(stat.st_mtime).isoformat()

            cursor.execute("""
                INSERT OR REPLACE INTO file_baselines 
                (filepath, sha256, permissions, file_size, last_modified)
                VALUES (?, ?, ?, ?, ?)
            """, (full_path, file_hash, perms, size, mtime))
            count += 1
            print(f"    [+] Registered {full_path} -> {file_hash[:12]}...")

    conn.commit()
    conn.close()
    print(f"[✓] Baseline established for {count} files.\n")

class FIMWatchdogHandler(FileSystemEventHandler):
    """Real-time File System Watcher responding to create, modify, delete events."""

    def on_modified(self, event):
        if event.is_directory:
            return
        filepath = os.path.abspath(event.src_path)
        new_hash = calculate_sha256(filepath)

        conn = get_db()
        cursor = conn.cursor()
        cursor.execute("SELECT sha256, permissions FROM file_baselines WHERE filepath = ?", (filepath,))
        row = cursor.fetchone()

        if row:
            old_hash = row["sha256"]
            if old_hash != new_hash and not new_hash.startswith("ERROR"):
                self.log_alert("FILE_MODIFIED", "CRITICAL", filepath, old_hash, new_hash,
                               "File contents altered unauthorizedly.")
        else:
            self.log_alert("UNTRACKED_MODIFICATION", "WARNING", filepath, None, new_hash,
                           "File modified without initial baseline record.")
        conn.close()

    def on_created(self, event):
        if event.is_directory:
            return
        filepath = os.path.abspath(event.src_path)
        new_hash = calculate_sha256(filepath)
        self.log_alert("FILE_CREATED", "HIGH", filepath, None, new_hash,
                       "New unauthorized file detected in monitored zone.")

    def on_deleted(self, event):
        if event.is_directory:
            return
        filepath = os.path.abspath(event.src_path)
        self.log_alert("FILE_DELETED", "CRITICAL", filepath, None, None,
                       "Monitored file was deleted or removed.")

    def log_alert(self, event_type, severity, filepath, old_hash, new_hash, details):
        conn = get_db()
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO security_alerts (event_type, severity, filepath, old_hash, new_hash, details)
            VALUES (?, ?, ?, ?, ?, ?)
        """, (event_type, severity, filepath, old_hash, new_hash, details))
        conn.commit()
        conn.close()

        print("\n" + "!" * 60)
        print(f"🚨 [SECURITY ALERT - {severity}] {event_type}")
        print(f"File:     {filepath}")
        print(f"Old Hash: {old_hash}")
        print(f"New Hash: {new_hash}")
        print(f"Details:  {details}")
        print("!" * 60 + "\n")

if __name__ == "__main__":
    init_database()
    watch_dir = sys.argv[1] if len(sys.argv) > 1 else "./monitored_folder"
    os.makedirs(watch_dir, exist_ok=True)
    
    establish_baseline(watch_dir)

    observer = Observer()
    observer.schedule(FIMWatchdogHandler(), path=watch_dir, recursive=True)
    observer.start()
    print(f"[*] FIM Watchdog active. Monitoring directory: {os.path.abspath(watch_dir)}")
    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        observer.stop()
        print("\n[-] AegisGuard FIM Daemon terminated.")
    observer.join()
